import axios from 'axios';
import AsyncStorage from '@react-native-async-storage/async-storage';

const API_URL = 'https://seu-backend.com/api'; // substitua pela URL real

// LOGIN com cpf ou cnpj, dependendo do tipo
const login = async (identificador, senha, tipo) => {
  const payload = tipo === 'pf'
    ? { cpf: identificador, senha }
    : { cnpj: identificador, senha };

  const response = await axios.post(`${API_URL}/login`, payload);
  const { token, usuario } = response.data;

  await AsyncStorage.setItem('token', token);
  return { token, usuario };
};

// Cadastro Pessoa Física
const cadastroPF = async (dados) => {
  return axios.post(`${API_URL}/cadastro/pf`, dados);
};

// Cadastro Pessoa Jurídica
const cadastroPJ = async (dados) => {
  return axios.post(`${API_URL}/cadastro/pj`, dados);
};

// Validação de dados para recuperação de senha
const recuperarSenha = async (dados) => {
  return axios.post(`${API_URL}/recuperar-senha`, dados);
};

// Redefinição de senha com dados e nova senha
const redefinirSenha = async (dados) => {
  return axios.post(`${API_URL}/redefinir-senha`, dados);
};

// Logout remove o token local
const logout = async () => {
  await AsyncStorage.removeItem('token');
};

// Recupera token salvo
const getToken = async () => {
  return await AsyncStorage.getItem('token');
};

export default {
  login,
  cadastroPF,
  cadastroPJ,
  recuperarSenha,
  redefinirSenha,
  logout,
  getToken,
};
